SOV_purge_escalate_effect = {
	if = { limit = { check_variable = { SOV_purge_level < 3 } }
		if = { limit = { has_idea = SOV_purge_spirit_1 } remove_ideas = SOV_purge_spirit_1 }
		else_if = { limit = { has_idea = SOV_purge_spirit_2 } remove_ideas = SOV_purge_spirit_2 }
		else_if = { limit = { has_idea = SOV_purge_spirit_3 } remove_ideas = SOV_purge_spirit_3 }
	}

	add_to_variable = { SOV_purge_level = 1 }
	clamp_variable = { var = SOV_purge_level min = 1 max = 3 }

	if = { limit = { check_variable = { SOV_purge_level = 1 } }
		add_timed_idea = { idea = SOV_purge_spirit_1 days = 900 } }
	else_if = { limit = { check_variable = { SOV_purge_level = 2 } }
		add_timed_idea = { idea = SOV_purge_spirit_2 days = 900 } }
	else = {
		add_timed_idea = { idea = SOV_purge_spirit_3 days = 900 } }
}

SOV_purge_stop_effect = {
	remove_ideas = SOV_purge_spirit_1
	remove_ideas = SOV_purge_spirit_2
	remove_ideas = SOV_purge_spirit_3
	set_variable = { SOV_purge_level = 0 }
}
